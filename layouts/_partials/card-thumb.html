{{/* card-thumb.html — the image URL for a page's card. The picture is chosen by card-image.html (cover, image URL, or the og:image of an external link) and resized to a 16:9 WebP so a 300 KB photo arrives as a few kilobytes. Returns an empty string when the page has no picture, or when the picture it names cannot be found; feed-card.html then draws a plain grey panel of the same shape in its place. */}} {{- $page := . -}} {{- $src := partial "card-image.html" $page -}} {{- $src = $src | replaceRE `^['"]+` "" | replaceRE `['"]+$` "" -}} {{- $url := "" -}} {{- $spec := "800x450 Smart webp q80" -}} {{- with $src -}} {{- $img := "" -}} {{- if or (hasPrefix . "http://") (hasPrefix . "https://") -}} {{- $opts := dict "headers" (dict "User-Agent" "Mozilla/5.0 (compatible; Hugo static site build)") -}} {{- with try (resources.GetRemote . $opts) -}} {{- with .Err }}{{ warnf "card image: could not fetch %q (%s)" $src . }} {{- else }}{{ with .Value }}{{ if eq .MediaType.MainType "image" }}{{ $img = . }}{{ end }}{{ end }}{{ end -}} {{- end -}} {{- else -}} {{- with $page.Resources.GetMatch . }}{{ $img = . }}{{ end -}} {{- if not $img -}} {{- /* static/img is also mounted as assets/img (see hugo.toml), so a path like /img/photo.jpg can be resized here. */ -}} {{- with resources.Get (strings.TrimPrefix "/" .) }}{{ $img = . }}{{ end -}} {{- end -}} {{- if not $img }}{{ warnf "card image: %q was not found in static/ or beside the post — check the path" $src }}{{ end -}} {{- end -}} {{- with $img -}} {{- if in (slice "svg" "gif") .MediaType.SubType -}} {{- $url = .RelPermalink -}} {{- else -}} {{- with try (.Fill $spec) }}{{ with .Value }}{{ $url = .RelPermalink }}{{ end }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $url -}}
